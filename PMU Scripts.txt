-------------------------------- GST  --------------------------------------------------

CREATE TABLE pmu.gst_master (
    id              BIGSERIAL PRIMARY KEY,
    code_type       CHARACTER VARYING(10) NOT NULL,      
    code            CHARACTER VARYING(10) NOT NULL,      
    description     TEXT NOT NULL,              
    gst_rate        NUMERIC(5,2) DEFAULT 0.00, 
    cgst_rate       NUMERIC(5,2) DEFAULT 0.00,
    sgst_rate       NUMERIC(5,2) DEFAULT 0.00,
    igst_rate       NUMERIC(5,2) DEFAULT 0.00,
    cess_rate       NUMERIC(5,2) DEFAULT 0.00,
	other_rate      NUMERIC(5,2) DEFAULT 0.00,
    effective_from  DATE NOT NULL,
    effective_to    DATE,
    is_active       BOOLEAN DEFAULT TRUE,
    created     	TIMESTAMP WITH TIME ZONE DEFAULT now(),
    updated_at      TIMESTAMP WITH TIME ZONE DEFAULT now(),
	UNIQUE (code_type, code),
	
	CONSTRAINT gst_master_effective_date_check
    CHECK (effective_to IS NULL OR effective_to >= effective_from)
);


-------------------------------- EMPANELMENT  --------------------------------------------------

CREATE TABLE pmu.empanelment_master (
    id                  BIGSERIAL PRIMARY KEY,

    category            CHARACTER VARYING(20) NOT NULL,    
    empanelment_no      CHARACTER VARYING(50) NOT NULL,
	amc					NUMERIC(5,2) DEFAULT 0.00,
	agm					NUMERIC(5,2) DEFAULT 0.00,

	rfe					CHARACTER VARYING(255) NOT NULL,

    effective_to        DATE,

	remarks				TEXT,

    updated_at      	TIMESTAMP WITH TIME ZONE DEFAULT now(),

    CHECK (effective_to IS NULL OR effective_to >= effective_from)

----------------------------------------------------------------- Employees  --------------------------------------------------------------------------------

CREATE TABLE pmu.employees (
    id BIGSERIAL PRIMARY KEY,

    -- System IDs
    employee_code VARCHAR(30) UNIQUE NOT NULL,
    attendance_code VARCHAR(20) UNIQUE,

    -- Authentication
    password_hash VARCHAR(255) NOT NULL,

    -- Name (minimal needed for UI + login)
    salutation VARCHAR(10) CHECK (salutation IN ('Mr','Ms','Mrs','Dr')),
    first_name VARCHAR(50) NOT NULL,
    middle_name VARCHAR(50),
    last_name VARCHAR(50) NOT NULL,

    -- Contact for login
    email VARCHAR(120) UNIQUE NOT NULL,
    contact_no VARCHAR(20) NOT NULL,

    -- Employment lifecycle
    date_of_joining DATE NOT NULL,
    probation_end_date DATE,
    confirmation_date DATE,
    date_of_exit DATE,
    notice_period_days INTEGER,

    -- HR workflow status
    account_status VARCHAR(20) CHECK (account_status IN ('PENDING','ACTIVE','REJECTED','EXITED')) DEFAULT 'PENDING', -- PENDING / ACTIVE / REJECTED / EXITED
    hr_verified BOOLEAN DEFAULT false,
    verified_by BIGINT REFERENCES pmu.employees(id),
    verified_at TIMESTAMP,

    -- Flags
    is_active BOOLEAN DEFAULT true,
    is_deleted BOOLEAN DEFAULT false,

    -- Audit
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_emp_email ON pmu.employees(email);
CREATE INDEX idx_emp_contact ON pmu.employees(contact_no);

----------------------------------------------------------------- Role Master ----------------------------------------------------------------- 

CREATE TABLE pmu.roles (
    id BIGSERIAL PRIMARY KEY,

    role_code VARCHAR(50) NOT NULL,   
    role_name VARCHAR(100) NOT NULL,
    description TEXT,

    is_system_role BOOLEAN DEFAULT false, -- cannot be deleted
	role_level INT NOT NULL, -- Lower number = higher authority
	is_super_admin BOOLEAN DEFAULT FALSE, -- bypass all permission checks
    is_active BOOLEAN DEFAULT TRUE,
    organization_id BIGINT, -- NULL = global role -- NOT NULL = org-specific role

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	
	UNIQUE (role_code, organization_id)
);

CREATE INDEX idx_roles_org ON pmu.roles(organization_id);
CREATE INDEX idx_roles_org_code ON pmu.roles(organization_id, role_code);

ALTER TABLE pmu.roles
ADD CONSTRAINT fk_roles_organization
FOREIGN KEY (organization_id)
REFERENCES pmu.organizations(id)
ON DELETE CASCADE
ON UPDATE CASCADE;

CREATE OR REPLACE FUNCTION pmu.fn_update_previous_salary_effective_to()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE pmu.employee_salary_structure
  SET effective_to = NEW.effective_from
  WHERE id = (
    SELECT id FROM pmu.employee_salary_structure
    WHERE employee_id = NEW.employee_id
      AND effective_from < NEW.effective_from
    ORDER BY effective_from DESC
    LIMIT 1
  );

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_previous_salary
BEFORE INSERT ON pmu.employee_salary_structure
FOR EACH ROW
EXECUTE FUNCTION pmu.fn_update_previous_salary_effective_to();

(
    id bigserial,

    name character varying(255) NOT NULL,

    code character varying(100) NOT NULL,

    description text,

    addon_type character varying(20) NOT NULL,
    -- EARNING / DEDUCTION

    taxable boolean DEFAULT true,

    affects_pf boolean DEFAULT false,

    affects_esi boolean DEFAULT false,

    recurring_allowed boolean DEFAULT false,

    is_active boolean DEFAULT true,

    is_deleted boolean DEFAULT false,

    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,

    )
);

CREATE TABLE IF NOT EXISTS pmu.wo_milestones
(
    id BIGSERIAL PRIMARY KEY,

    work_order_id BIGINT NOT NULL
        REFERENCES pmu.work_order(id)
        ON DELETE CASCADE,

    gst_code_fk BIGINT NOT NULL
        REFERENCES pmu.gst_master(id)
        ON DELETE CASCADE,

    title CHARACTER VARYING(255) NOT NULL,

    description TEXT,

    amount NUMERIC(15,2) NOT NULL,

    from_date DATE,
    to_date DATE,

    is_active BOOLEAN DEFAULT TRUE,
    created_on TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    modified_on TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    created_by BIGINT,

    CONSTRAINT wo_milestones_date_check
    CHECK (
        to_date IS NULL
        OR from_date IS NULL
        OR to_date >= from_date
    )
);
